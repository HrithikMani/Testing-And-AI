cmd = 'Close Popup'
if OKRA_TEXT:
	# Try to find the miewin by title
	driver.clickElement(xpath='''//div[@class="wc_win"][last()]//div[contains(@class, 'wc_win_title') and contains(., '{OKRA_TEXT}')]/parent::div/*[self::input or self::button][contains(@class, 'xclose')]|//div[contains(@class, "modal")]//div[contains(., "{OKRA_TEXT}")]''')
else:
	# Just close the most recent one by hitting ESCAPE
	wd = driver.getWebDriver()
	from selenium.webdriver.common.action_chains import ActionChains
	from selenium.webdriver.common.keys import Keys
	modal = '//div[contains(@class, "wc_win") or contains(@class, "modal") or contains(@class, "miewin") or contains(@class, "growl")]'
	if driver.waitFor(driver, lambda d: d.getElement(xpath=modal), expected_return=True, timeout=60):
		driver.pause(10)
		ActionChains(wd).send_keys(Keys.ESCAPE).perform()
		driver.reportCommandStatus(cmd, 'Escape key sent', True, '', '')
	else:
		driver.reportCommandStatus(cmd, 'Modal element not found', False, '', '')
