def qlopHoverClick():
	oldtimeout = driver.timeout
	driver.setTimeout(5)
	driver.reportCommandStatus('', '', True, 'section xpath', section)
	driver.reportCommandStatus('', '', True, 'qlitem xpath', qlitem)
	driver.reportCommandStatus('', '', True, 'qlop xpath', qlop)
	if not driver.wcutils.waitFor(lambda d: d.runJS('return wcutils._layoutOps === 0 && miehttp.isWaiting() === false'), timeout=60,
		comments='Wait for ajax and layoutInsert operations to complete prior to adding'):
		driver.reportCommandStatus('LayoutOps', driver.runJS('return wcutils._layoutOps'), None, 'isWaiting()', driver.runJS('return miehttp.isWaiting()'))
	driver.wcutils.waitForAJAX(timeout=60, quiet=5)
	if driver.getElement(xpath=section):
		if driver.waitFor(driver, lambda driver: driver.getElement(xpath=qlitem), expected_return=True):
			driver.runJS('arguments[0].scrollIntoView(true)', driver.getElement(xpath=section))
			driver.mouseOver(xpath=qlitem)
		else:
			driver.reportCommandStatus('', '', False, '', 'Quicklist dynItem not found')
	else:
		driver.reportCommandStatus('', '', False, '', 'Section not found')
	if driver.waitFor(driver, lambda driver: driver.getElement(xpath=qlitem), expected_return=True):
		driver.runJS('arguments[0].scrollIntoView(true)', driver.getElement(xpath=section))
		driver.mouseOver(xpath=qlitem)
		if driver.waitFor(driver, lambda driver: driver.getElement(xpath=qlop), expected_return=True):
			driver.clickElement(xpath=qlop)
		elif not driver.waitFor(driver, lambda driver: driver.getElement(xpath=qlop), expected_return=True):
			driver.reportCommandStatus('', '', True, '', '2nd attempt')
			driver.runJS('arguments[0].scrollIntoView(true)', driver.getElement(xpath=section))
			driver.mouseOver(xpath=qlitem)
			if driver.waitFor(driver, lambda driver: driver.getElement(xpath=qlop), expected_return=True):
				driver.clickElement(xpath=qlop)
			elif not driver.waitFor(driver, lambda driver: driver.getElement(xpath=qlop), expected_return=True, timeout=60):
				driver.reportCommandStatus('', '', True, '', '3rd attempt')
				driver.wcutils.waitForAJAX(timeout=30)
				driver.runJS('arguments[0].scrollIntoView(true)', driver.getElement(xpath=section))
				driver.mouseOver(xpath=qlitem)
				driver.waitFor(driver, lambda driver: driver.getElement(xpath=qlop), expected_return=True, timeout=60)
				driver.clickElement(xpath=qlop)
			else:
				driver.reportCommandStatus('quicklist.pod', 'Edit/Correct Medication/Change Therapy', False, 'Variables given', 'Unable to click DynOps button after 2 attempts')
		else:
			driver.reportCommandStatus('quicklist.pod', 'Edit/Correct Medication/Change Therapy', False, 'Variables given', 'DynOps button not found')
	else:
		driver.reportCommandStatus('quicklist.pod', 'Edit/Correct Medication/Change Therapy', False, 'Variables given', 'Quicklist Item not found')
	if driver.getElement(xpath='//div[@id="miewin_progress"]'):
		driver.wcutils.waitFor(lambda d: d.getElement(xpath='//div[@id="miewin_progress"]'), timeout=60, expected=False, comments='Wait for miewin to finish building')
	if not driver.wcutils.waitFor(lambda d: d.runJS('return wcutils._layoutOps === 0 && miehttp.isWaiting() === false'), timeout=60,
		comments='Wait for ajax and layoutInsert operations to complete after inserting item'):
		driver.reportCommandStatus('LayoutOps', driver.runJS('return wcutils._layoutOps'), None, 'isWaiting()', driver.runJS('return miehttp.isWaiting()'))
	driver.setTimeout(oldtimeout)
