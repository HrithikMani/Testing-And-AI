menuopen = '//body[contains(@class, "wc_slideInSideMenu") and not(contains(@class, "wc_sidebarPinned"))]'
menupinned = '//body[contains(@class, "wc_sidebarPinned")]'
menuclosed = '//body[not(contains(@class, "wc_slideInSideMenu")) and not(contains(@class, " wc_sidebarPinned"))]'
if driver.wcutils.waitFor(lambda d: d.getElement(xpath=menuclosed):
	driver.reportCommandStatus('sidemenu.pod', 'Close Verb', True, '', 'Sidemenu is already closed')
else:
	if driver.wcutils.waitFor(lambda d: d.getElement(xpath=menuopen):
		driver.wcutils.waitFor(lambda d: d.getElement(xpath=menuclosed)
		driver.reportCommandStatus('sidemenu.pod', 'Close Verb', True, '', 'Sidemenu has been closed')
	elif driver.wcutils.waitFor(lambda d: d.getElement(xpath=menupinned):
		driver.clickElement(xpath='//span[@id="thumbtack"]')
		driver.wcutils.waitFor(lambda d: d.getElement(xpath=menuclosed)
		driver.reportCommandStatus('sidemenu.pod', 'Close Verb', True, '', 'Sidemenu has been closed')
	else:
		driver.reportCommandStatus('sidemenu.pod', 'Close Verb', False, '', 'Something went wrong')
