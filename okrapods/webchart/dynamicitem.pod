def dynopHoverClick():
	oldtimeout = driver.timeout
	driver.setTimeout(5)
	driver.reportCommandStatus('', '', True, 'section xpath', section)
	driver.reportCommandStatus('', '', True, 'dynitem xpath', dynitem)
	driver.reportCommandStatus('', '', True, 'dynop xpath', dynop)
	if not driver.wcutils.waitFor(lambda d: d.runJS('return wcutils._layoutOps === 0 && miehttp.isWaiting() === false'), timeout=60,
		comments='Wait for ajax and layoutInsert operations to complete prior to adding'):
		driver.reportCommandStatus('LayoutOps', driver.runJS('return wcutils._layoutOps'), None, 'isWaiting()', driver.runJS('return miehttp.isWaiting()'))
	driver.wcutils.waitForAJAX(timeout=60, quiet=5)
	if driver.getElement(xpath=section):
		if driver.waitFor(driver, lambda driver: driver.getElement(xpath=dynitem), expected_return=True):
			driver.runJS('arguments[0].scrollIntoView(true)', driver.getElement(xpath=section))
			driver.mouseOver(xpath=dynitem)
		else:
			driver.reportCommandStatus('', '', False, '', 'Dynamic dynItem not found')
	else:
		driver.reportCommandStatus('', '', False, '', 'Section not found')
	if driver.waitFor(driver, lambda driver: driver.getElement(xpath=dynitem), expected_return=True):
		driver.reportCommandStatus('', '', True, '', '1st attempt')
		driver.runJS('arguments[0].scrollIntoView(true)', driver.getElement(xpath=section))
		driver.mouseOver(xpath=dynitem)
		if driver.waitFor(driver, lambda driver: driver.getElement(xpath=dynop), expected_return=True):
			driver.clickElement(xpath=dynop)
		elif not driver.waitFor(driver, lambda driver: driver.getElement(xpath=dynop), expected_return=True):
			driver.reportCommandStatus('', '', True, '', '2nd attempt')
			driver.runJS('arguments[0].scrollIntoView(true)', driver.getElement(xpath=section))
			driver.mouseOver(xpath=dynitem)
			if driver.waitFor(driver, lambda driver: driver.getElement(xpath=dynop), expected_return=True):
				driver.clickElement(xpath=dynop)
			elif not driver.waitFor(driver, lambda driver: driver.getElement(xpath=dynop), expected_return=True, timeout=60):
				driver.reportCommandStatus('', '', True, '', '3rd attempt')
				driver.wcutils.waitForAJAX(timeout=30)
				driver.runJS('arguments[0].scrollIntoView(true)', driver.getElement(xpath=section))
				driver.mouseOver(xpath=dynitem)
				driver.waitFor(driver, lambda driver: driver.getElement(xpath=dynop), expected_return=True, timeout=60)
				driver.clickElement(xpath=dynop)
			else:
				driver.reportCommandStatus('dynamicitem.pod', 'Edit/Correct Medication/Change Therapy', False, 'Variables given', 'Unable to click DynOps button after 2 attempts')
		else:
			driver.reportCommandStatus('dynamicitem.pod', 'Edit/Correct Medication/Change Therapy', False, 'Variables given', 'DynOps button not found')
	else:
		driver.reportCommandStatus('dynamicitem.pod', 'Edit/Correct Medication/Change Therapy', False, 'Variables given', 'Quicklist Item not found')
	if driver.getElement(xpath='//div[@id="miewin_progress"]'):
		driver.wcutils.waitFor(lambda d: d.getElement(xpath='//div[@id="miewin_progress"]'), timeout=60, expected=False, comments='Wait for miewin to finish building')
	if not driver.wcutils.waitFor(lambda d: d.runJS('return wcutils._layoutOps === 0 && miehttp.isWaiting() === false'), timeout=60,
		comments='Wait for ajax and layoutInsert operations to complete after inserting item'):
		driver.reportCommandStatus('LayoutOps', driver.runJS('return wcutils._layoutOps'), None, 'isWaiting()', driver.runJS('return miehttp.isWaiting()'))
	driver.setTimeout(oldtimeout)
