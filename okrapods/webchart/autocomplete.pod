def executeAutocomplete():
	oldtimeout = driver.timeout
	driver.setTimeout(10)
	if driver.wcutils.waitForEle(xpath=acxpath, timeout=30):
		driver.runJS('arguments[0].scrollIntoView()', driver.getElement(xpath=acxpath))
	else:
		driver.reportCommandStatus('autocomplete.pod', '', False, '', 'Autocomplete not present, not scrolled into view')
	if driver.wcutils.waitForEle(xpath=acxpath, timeout=30):
		driver.wcutils.waitForAJAX(timeout=60, quiet=10)
		driver.enterFormData(shortstr, xpath=acxpath, clear=True, blur=False)
		if not driver.wcutils.waitFor(lambda d: d.getElement(xpath=pending), expected=False, timeout=30):
			driver.reportCommandStatus('autocomplete.pod', '', False, '', 'Autocomplete options pending after 30s')
		if driver.getElement(xpath=full):
			driver.clickElement(xpath=full)
		elif not driver.getElement(xpath=full):
			driver.enterFormData(fullstr, xpath=acxpath, clear=True, blur=False)
			if not driver.wcutils.waitFor(lambda d: d.getElement(xpath=pending), expected=False, timeout=30):
				driver.reportCommandStatus('autocomplete.pod', '', False, '', 'Autocomplete options pending after 30s')
			driver.wcutils.waitFor(lambda d: d.getElement(xpath=full))
			if driver.getElement(xpath=full):
				driver.clickElement(xpath=full)
			else:
				driver.reportCommandStatus('autocomplete.pod', '', False, '', 'Autocomplete options not found when using the full string')
		else:
			driver.reportCommandStatus('autocomplete.pod', '', False, '', 'Autocomplete options not found when using the short string')
	else:
		driver.reportCommandStatus('autocomplete.pod', '', False, '', 'Autocomplete input not found')
	driver.setTimeout(oldtimeout)
